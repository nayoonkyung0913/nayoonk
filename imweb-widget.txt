<script>
(function () {
  var key = Symbol.for("mines.imweb.fx.loader.v1");
  if (window[key]) return;
  var selector = '[data-imweb-fx="1"]';
  var src = "https://nayoonkyung0913.github.io/nayoonk/imweb-effects.js";
  var pending = false;
  var watcher = new MutationObserver(function (records) {
    if (pending) return;
    for (var r of records) {
      if (r.type === "attributes" && r.target.matches(selector)) return load();
      for (var el of r.addedNodes) {
        if (el.nodeType === 1 && (el.matches(selector) || el.querySelector(selector))) return load();
      }
    }
  });
  function load() {
    if (pending || !document.querySelector(selector)) return;
    pending = true;
    var tag = document.createElement("script");
    tag.src = src;
    tag.onload = function () { watcher.disconnect(); };
    tag.onerror = function () { pending = false; tag.remove(); };
    document.head.appendChild(tag);
  }
  window[key] = true;
  watcher.observe(document.documentElement, {childList: true, subtree: true, attributes: true, attributeFilter: ["data-imweb-fx"]});
  load();
})();
</script>
